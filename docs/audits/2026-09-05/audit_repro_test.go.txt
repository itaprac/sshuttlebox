func auditSetup(t *testing.T) (string, config.Config) {
 t.Helper()
 home := withTempHome(t)
 t.Setenv("XDG_STATE_HOME", filepath.Join(home, "state"))
 path, _, err := config.Init(); if err != nil { t.Fatal(err) }
 cfg := config.Default()
 cfg.Hosts["prod"] = config.Host{Host:"example.invalid"}
 cfg.Tunnels["db"] = config.Tunnel{Host:"prod", Type:"local", LocalPort:15432, RemoteHost:"localhost", RemotePort:5432}
 if err := config.Save(path,cfg); err != nil {t.Fatal(err)}
 return path,cfg
}

func TestAuditRenderProbesInvisibleRows(t *testing.T) {
 path,cfg:=auditSetup(t)
 logPath:=filepath.Join(t.TempDir(),"calls")
 t.Setenv("AUDIT_CALL_LOG",logPath)
 bin:=filepath.Join(t.TempDir(),"ssh")
 if err:=os.WriteFile(bin,[]byte("#!/bin/sh\necho check >> \"$AUDIT_CALL_LOG\"\nsleep 0.01\nexit 0\n"),0700);err!=nil {t.Fatal(err)}
 t.Setenv("SHBX_SSH_BIN",bin)
 state:=tunnelstate.Default();cfg.Tunnels=map[string]config.Tunnel{}
 for i:=0;i<40;i++ {name:=fmt.Sprintf("tunnel-%02d",i);cfg.Tunnels[name]=config.Tunnel{Host:"prod",Type:"dynamic",LocalPort:1080+i};state.Tunnels[name]=tunnelstate.NewEntry(99999999,"ssh example.invalid","/tmp/audit-fictional.sock")}
 if err:=tunnelstate.Save(state);err!=nil {t.Fatal(err)}
 m:=newTUIModelWithState(path,cfg,nil)
 start:=time.Now();view:=m.tunnelsPaneView(40,true,8);elapsed:=time.Since(start)
 data,err:=os.ReadFile(logPath);if err!=nil {t.Fatal(err)}
 count:=strings.Count(string(data),"check")
 if count!=40 {t.Fatalf("unexpected checks: %d",count)}
 t.Logf("CONFIRMED: %d synchronous SSH probes for %d visible lines; %s with 10 ms simulated probe delay",count,len(strings.Split(view,"\n")),elapsed)
}

func TestAuditSFTPPasswordBatchDisablesPasswordAuth(t *testing.T) {
 auditSetup(t)
 if _,err:=os.Stat("/usr/bin/sftp");err!=nil {t.Skip(err)}
 dir:=t.TempDir();capture:=filepath.Join(dir,"args");transport:=filepath.Join(dir,"ssh");wrapper:=filepath.Join(dir,"sftp")
 t.Setenv("AUDIT_ARGS",capture);t.Setenv("AUDIT_TRANSPORT",transport)
 if err:=os.WriteFile(transport,[]byte("#!/bin/sh\nprintf '%s\\n' \"$@\" > \"$AUDIT_ARGS\"\nexit 1\n"),0700);err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(wrapper,[]byte("#!/bin/sh\nexec /usr/bin/sftp -S \"$AUDIT_TRANSPORT\" \"$@\"\n"),0700);err!=nil {t.Fatal(err)}
 t.Setenv("SHBX_SFTP_BIN",wrapper)
 _=runSFTPProcess(config.Host{Host:"example.invalid",Password:"dummy"},buildSFTPArgs(config.Host{Host:"example.invalid"},""),"ls\n")
 args,err:=os.ReadFile(capture);if err!=nil {t.Fatal(err)}
 if !strings.Contains(strings.ToLower(string(args)),"batchmode yes") {t.Fatalf("unexpected args: %s",args)}
 t.Log("CONFIRMED using system sftp: password batch invokes SSH with -oBatchMode=yes")
}

func TestAuditPasswordInjectedIntoApplicationPrompt(t *testing.T) {
 master,slave,err:=pty.Open();if err!=nil {t.Fatal(err)}
 defer master.Close();defer slave.Close()
 _,err=term.MakeRaw(int(slave.Fd()));if err!=nil {t.Fatal(err)}
 done:=make(chan error,1)
 go func(){done<-copySSHOutputAndInjectPassword(io.Discard,master,"dummy-audit-secret")}()
 if _,err:=slave.Write([]byte("Authenticated with public key.\r\nApplication password:"));err!=nil {t.Fatal(err)}
 read:=make(chan []byte,1)
 go func(){b:=make([]byte,128);n,_:=slave.Read(b);read<-b[:n]}()
 select {case b:=<-read:if !bytes.Contains(b,[]byte("dummy-audit-secret")){t.Fatal("bug not reproduced")};t.Log("CONFIRMED: stored SSH password is injected into post-login application input")
 case <-time.After(2*time.Second):t.Fatal("timed out")}
 master.Close();slave.Close();<-done
}

func TestAuditTypingQQuitsForms(t *testing.T) {
 path,cfg:=auditSetup(t)
 for _,screen:=range []string{"host","tunnel"} {
 m:=newTUIModelWithState(path,cfg,nil)
 var model tea.Model
 if screen=="host" {model,_=m.openForm("",config.Host{})} else {model,_=m.openTunnelForm("",config.Tunnel{Type:"local"})}
 model,cmd:=model.Update(tea.KeyMsg{Type:tea.KeyRunes,Runes:[]rune{'q'}})
 if cmd==nil {t.Fatal("bug not reproduced")}
 if _,ok:=cmd().(tea.QuitMsg);!ok {t.Fatal("not quit command")}
 t.Logf("CONFIRMED: typing q in %s form quits app",screen)
 }
}
